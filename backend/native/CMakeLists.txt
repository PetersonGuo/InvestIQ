cmake_minimum_required(VERSION 3.20)
project(StockAssistNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_package(Python3 COMPONENTS Interpreter Development.Embed REQUIRED)
set(IBKR_CPP_SDK_DIR "$ENV{IBKR_CPP_SDK_DIR}" CACHE PATH "Existing licensed IBKR C++ client source directory")
if(NOT IBKR_CPP_SDK_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../Quant/client/EClientSocket.h")
  set(IBKR_CPP_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../Quant/client")
endif()
if(NOT EXISTS "${IBKR_CPP_SDK_DIR}/EClientSocket.h")
  message(FATAL_ERROR "Set IBKR_CPP_SDK_DIR to your installed IBKR C++ client source directory (containing EClientSocket.h and proto/).")
endif()
find_package(Protobuf CONFIG REQUIRED)
file(GLOB PROTOS CONFIGURE_DEPENDS "${IBKR_CPP_SDK_DIR}/proto/*.proto")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/sdk/protobufUnix")
foreach(P ${PROTOS})
  get_filename_component(N ${P} NAME_WE)
  list(APPEND PROTO_SOURCES "${CMAKE_BINARY_DIR}/sdk/protobufUnix/${N}.pb.cc")
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/sdk/protobufUnix/${N}.pb.cc" "${CMAKE_BINARY_DIR}/sdk/protobufUnix/${N}.pb.h"
    COMMAND protobuf::protoc "--proto_path=${IBKR_CPP_SDK_DIR}/proto" "--cpp_out=${CMAKE_BINARY_DIR}/sdk/protobufUnix" "${P}" DEPENDS "${P}" VERBATIM)
endforeach()
add_library(ibkr_proto STATIC ${PROTO_SOURCES})
target_include_directories(ibkr_proto PUBLIC "${CMAKE_BINARY_DIR}/sdk" "${CMAKE_BINARY_DIR}/sdk/protobufUnix")
target_link_libraries(ibkr_proto PUBLIC protobuf::libprotobuf)
file(GLOB SDK_SOURCES CONFIGURE_DEPENDS "${IBKR_CPP_SDK_DIR}/*.cpp")
add_library(ibkr_sdk STATIC ${SDK_SOURCES})
target_include_directories(ibkr_sdk PUBLIC "${IBKR_CPP_SDK_DIR}")
target_link_libraries(ibkr_sdk PUBLIC ibkr_proto Threads::Threads)
file(GLOB OWN_SOURCES CONFIGURE_DEPENDS src/*.cpp src/*.hpp ../simulation/strategy_api.h ../simulation/native_engine.cpp ../simulation/python_strategy.py)
set(FINGERPRINT "native-v3|${CMAKE_CXX_COMPILER_ID}|${CMAKE_CXX_COMPILER_VERSION}|${Python3_VERSION}")
foreach(F ${OWN_SOURCES})
  file(SHA256 "${F}" HASH)
  string(APPEND FINGERPRINT "${HASH}")
endforeach()
string(SHA256 BUILD_ID "${FINGERPRINT}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${OWN_SOURCES})
configure_file(build_config.hpp.in build_config.hpp @ONLY)
add_library(sa_common STATIC src/common.cpp)
target_include_directories(sa_common PUBLIC vendor src "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sa_common PUBLIC SQLite3::SQLite3 OpenSSL::Crypto Threads::Threads)
add_executable(stockassist-server src/server.cpp src/market.cpp src/ibkr.cpp src/pairs.cpp src/backtests.cpp)
target_link_libraries(stockassist-server PRIVATE sa_common ibkr_sdk CURL::libcurl)
add_executable(stockassist-worker src/worker.cpp src/simulation.cpp ../simulation/native_engine.cpp)
target_include_directories(stockassist-worker PRIVATE ../simulation)
target_link_libraries(stockassist-worker PRIVATE sa_common ${CMAKE_DL_LIBS})
add_library(stockassist-python-adapter MODULE src/python_adapter.cpp)
set_target_properties(stockassist-python-adapter PROPERTIES PREFIX "" SUFFIX ".so")
target_include_directories(stockassist-python-adapter PRIVATE ../simulation "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(stockassist-python-adapter PRIVATE Python3::Python)
target_compile_definitions(stockassist-python-adapter PRIVATE SA_PYTHON_EXECUTABLE="${Python3_EXECUTABLE}")
enable_testing()
add_executable(stockassist-unit ../tests/native/unit.cpp src/simulation.cpp src/pairs.cpp src/market.cpp src/ibkr.cpp ../simulation/native_engine.cpp)
target_include_directories(stockassist-unit PRIVATE ../simulation)
target_link_libraries(stockassist-unit PRIVATE sa_common ibkr_sdk CURL::libcurl)
add_test(NAME native-unit COMMAND stockassist-unit)
